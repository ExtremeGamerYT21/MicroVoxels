cmake_minimum_required(VERSION 3.24)
project(Microvoxels VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MICROVOXELS_BUILD_DEMO "Build the Vulkan desktop demo" ON)
include(CTest)
add_executable(lattice_tests tests/lattice_tests.cpp)
target_include_directories(lattice_tests PRIVATE src)
add_test(NAME lattice COMMAND lattice_tests)
find_package(glm CONFIG REQUIRED)
add_executable(footprint_tests tests/footprint_tests.cpp)
target_include_directories(footprint_tests PRIVATE src)
target_link_libraries(footprint_tests PRIVATE glm::glm)
add_test(NAME surface_footprint COMMAND footprint_tests)
if(MICROVOXELS_BUILD_DEMO)
  find_package(Vulkan REQUIRED COMPONENTS glslangValidator)
  find_package(glfw3 3.3 REQUIRED)
  set(SHADER_FILES source.vert source.frag shadow.vert voxel.vert voxel.frag voxel_debug.vert voxel_debug.frag voxel_indexed.vert voxel_faces.vert voxel_faces_debug.vert fullscreen.vert fullscreen.frag hud.vert hud.frag lod.comp lod_resolve.comp voxelize.comp compact.comp compact_counts.comp compact_prefix.comp source_animate.comp)
  foreach(SHADER IN LISTS SHADER_FILES)
    set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv")
    add_custom_command(OUTPUT "${OUTPUT}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
      COMMAND Vulkan::glslangValidator -V --target-env vulkan1.1 -I${CMAKE_CURRENT_SOURCE_DIR}/shaders "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" -o "${OUTPUT}"
      DEPENDS "shaders/${SHADER}" shaders/common.glsl shaders/lattice.glsl shaders/cube.glsl shaders/cube_draw.glsl shaders/source_vertex.glsl shaders/source_shading.glsl shaders/surface_footprint.glsl shaders/counters.glsl VERBATIM)
    list(APPEND SHADER_OUTPUTS "${OUTPUT}")
  endforeach()
  add_custom_target(microvoxels_shaders ALL DEPENDS ${SHADER_OUTPUTS})
  add_executable(microvoxels src/main.cpp src/vulkan_context.cpp src/source_world.cpp src/source_renderer.cpp src/visual_voxelizer.cpp src/voxel_renderer.cpp src/hud.cpp)
  target_link_libraries(microvoxels PRIVATE Vulkan::Vulkan glfw glm::glm)
  target_compile_definitions(microvoxels PRIVATE GLM_FORCE_RADIANS GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_CTOR_INIT MICROVOXELS_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
  if(MSVC)
    target_compile_options(microvoxels PRIVATE /W4 /permissive-)
  else()
    target_compile_options(microvoxels PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
  endif()
  add_dependencies(microvoxels microvoxels_shaders)
endif()
